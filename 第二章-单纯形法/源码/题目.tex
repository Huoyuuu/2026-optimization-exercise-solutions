\section*{第一部分：考试真题（两阶段单纯形法）}
\begin{enumerate}
    \item $\min\ -x_1 - 2x_2 + x_3 \quad \text{s.t.} \begin{cases} x_1 - x_2 + x_3 \ge 2 \\ x_1 + x_2 \ge 3 \\ x_2 \ge 1 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\min\ 5x_1 + 3x_2 + 2x_3 \quad \text{s.t.} \begin{cases} 3x_1 + 2x_2 + x_3 \ge 2 \\ x_1 - x_2 + 2x_3 \ge 3 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\min\ 3x_1 - x_3 \quad \text{s.t.} \begin{cases} x_1 + x_2 + x_3 \le 6 \\ -2x_1 + x_2 - x_3 \ge 1 \\ 3x_2 + x_3 = 9 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\min\ 2x_1 + 3x_2 + 4x_3 \quad \text{s.t.} \begin{cases} x_1 + 2x_2 + x_3 \ge 3 \\ 2x_1 - x_2 + 3x_3 \ge 4 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\min\ -2x_1 + 3x_2 - x_3 \quad \text{s.t.} \begin{cases} x_1 - 2x_2 + x_3 \le 2 \\ 2x_1 - x_2 - x_3 \le -3 \\ 2x_1 - x_3 = 1 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
\end{enumerate}

\section*{第二部分：课后习题}
\subsection*{2.17 单纯形法}
\begin{enumerate}
    \item $\min\ x_1 - 3x_2 + 2x_3 \quad \text{s.t.} \begin{cases} -2x_1 + 4x_2 \le 12 \\ 3x_1 - x_2 + 2x_3 \le 7 \\ -4x_1 + 3x_2 + 8x_3 \le 10 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\max\ x_1 + x_2 + x_3 \quad \text{s.t.} \begin{cases} -x_1 - 2x_3 \le 5 \\ 2x_1 - 3x_2 + x_3 \le 3 \\ 2x_1 - 5x_2 + 6x_3 \le 5 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\max\ x_1 - x_2 + 3x_3 \quad \text{s.t.} \begin{cases} 2x_1 - 2x_2 - 3x_3 \le 0 \\ 2x_2 - x_3 \le 2 \\ x_1 + x_2 + x_3 \le 10 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\min\ -x_1 + 3x_2 - 3x_3 \quad \text{s.t.} \begin{cases} 3x_1 + x_2 + 2x_3 + x_5 = 5 \\ x_1 + x_3 + 2x_5 + x_6 = 2 \\ x_1 + 2x_3 + x_4 + 2x_5 = 6 \\ x_1, \dots, x_6 \ge 0 \end{cases}$
\end{enumerate}

\subsection*{2.18 两阶段单纯形法}
\begin{enumerate}
    \item $\min\ x_1 - x_2 + x_3 \quad \text{s.t.} \begin{cases} x_1 + 2x_2 + 3x_3 = 6 \\ 4x_1 + 5x_2 - 6x_3 = 6 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\min\ x_1 - x_2 + x_3 \quad \text{s.t.} \begin{cases} x_1 + 2x_2 + 3x_3 = 6 \\ 4x_1 + 5x_2 - 6x_3 = 6 \\ 6x_1 + 9x_2 = 18 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\min\ x_1 + 2x_2 + x_3 \quad \text{s.t.} \begin{cases} x_1 + 3x_2 + x_3 \ge 4 \\ x_1 + 2x_2 - x_3 \ge 6 \\ x_1 + x_3 \le 12 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\min\ 4x_1 + 5x_2 + 6x_3 \quad \text{s.t.} \begin{cases} x_1 + x_2 + x_3 = 5 \\ -6x_1 + 10x_2 + 5x_3 \le 20 \\ 5x_1 - 3x_2 + x_3 \ge 15 \\ x_1, x_2, x_3 \ge 0 \end{cases}$
    \item $\min\ 2x_1 + 2x_2 + 3x_3 + x_4 \quad \text{s.t.} \begin{cases} 4x_1 + x_2 - 2x_3 - x_4 = 4 \\ 10x_1 + 3x_2 - 7x_3 - 4x_4 = 12 \\ 14x_1 + 4x_2 - 9x_3 - 5x_4 = 16 \\ x_1, x_2, x_3, x_4 \ge 0 \end{cases}$
\end{enumerate}
